cmake_minimum_required(VERSION 3.10)
project(c_perf_benchmarks C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

find_package(OpenMP)

add_executable(cpu_prime_baseline src/cpu_prime_baseline.c src/common.c)
add_executable(cpu_prime_optimized src/cpu_prime_optimized.c src/common.c)
add_executable(memory_matrix_baseline src/memory_matrix_baseline.c src/common.c)
add_executable(memory_matrix_optimized src/memory_matrix_optimized.c src/common.c)
add_executable(file_io_benchmark src/file_io_benchmark.c src/common.c)
add_executable(mixed_sort_baseline src/mixed_sort_baseline.c src/common.c)
add_executable(mixed_sort_optimized src/mixed_sort_optimized.c src/common.c)

if(OpenMP_C_FOUND)
  target_link_libraries(cpu_prime_optimized PUBLIC OpenMP::OpenMP_C)
  target_link_libraries(memory_matrix_optimized PUBLIC OpenMP::OpenMP_C)
  target_link_libraries(mixed_sort_optimized PUBLIC OpenMP::OpenMP_C)
endif()

if(UNIX)
  target_link_libraries(cpu_prime_baseline m)
  target_link_libraries(cpu_prime_optimized m)
  target_link_libraries(memory_matrix_baseline m)
  target_link_libraries(memory_matrix_optimized m)
  target_link_libraries(file_io_benchmark m)
  target_link_libraries(mixed_sort_baseline m)
  target_link_libraries(mixed_sort_optimized m)
endif()
